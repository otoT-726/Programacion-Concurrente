sem mutex = 1, limiteT = 5, limiteM = 5, limiteC = 7, carga = 0;
sem espera[T + M] = ([T + M] 0);
Cola C;

process CamionTrigo[id: 0..T-1]{
  P(limiteT);
  P(limiteC);
  P(mutex);
  push(C, id);
  V(mutex);
  V(carga);
  P(espera[id]);
  V(limiteC);
  V(limiteT);
}

process CamionMaiz[id: T.. (T+M)-1]{
  P(limiteM);
  P(limiteC);
  P(mutex);
  push(C, id);
  V(mutex);
  V(carga);
  P(espera[id]);
  V(limiteC);
  V(limiteM);
}

process Coordinador{
  int aux;
  for(int i = 0 to M + T - 1){
    P(carga);
    P(mutex);
    pop(C, aux);
    V(mutex);
    V(espera[aux]);
  }
}
